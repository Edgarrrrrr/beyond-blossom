cmake_minimum_required(VERSION 3.21)
project(beyond_blossom_ae LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
set(BB_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Beyond Blossom source checkout")
set(BB_BUILD_TESTS OFF CACHE BOOL "AE native build" FORCE)
set(BB_BUILD_PYTHON OFF CACHE BOOL "AE native build" FORCE)
set(BB_BUILD_PROFILE_TOOLS OFF CACHE BOOL "Pure timing" FORCE)
set(BB_PREDECODE_SIMD ON CACHE BOOL "Adopted radius scan" FORCE)
set(BB_STRUCTURED_STORAGE_DIRECTIONS 12 CACHE STRING "Paper layout" FORCE)
set(BB_SHARED_COMPONENT_PARALLELISM 2 CACHE STRING "Independent components" FORCE)
set(BB_STRUCTURED_PROFILE_COUNTERS OFF CACHE BOOL "Pure timing" FORCE)
set(BB_STRUCTURED_PROFILE_TIMERS OFF CACHE BOOL "Pure timing" FORCE)
set(BB_STRUCTURED_PROFILE_TIMELINE OFF CACHE BOOL "Pure timing" FORCE)
set(BB_TAILRESET_PREDECODE_MODE 0 CACHE STRING "0 degree, 1 skip, 2 coordinate, 3 simple equal-share")
set(BB_TAILRESET_SYNDROME_STATS OFF CACHE BOOL "Separate untimed residual replay")
set(BB_AE_CACHE_LEVEL 3 CACHE STRING "Cache events: 2 or 3, Xeon only")
add_subdirectory("${BB_SOURCE_DIR}" decoder EXCLUDE_FROM_ALL)
set(BB_AE_STIM_ARCHIVE "" CACHE FILEPATH "Optional prebuilt Stim archive from the fixed dependency source")
if(BB_AE_STIM_ARCHIVE)
  if(NOT EXISTS "${BB_AE_STIM_ARCHIVE}" OR NOT IS_DIRECTORY "${FETCHCONTENT_SOURCE_DIR_STIM}/src")
    message(FATAL_ERROR "A prebuilt Stim archive requires its matching local source")
  endif()
  add_library(ae_stim STATIC IMPORTED GLOBAL)
  set_target_properties(ae_stim PROPERTIES IMPORTED_LOCATION "${BB_AE_STIM_ARCHIVE}"
    INTERFACE_INCLUDE_DIRECTORIES "${FETCHCONTENT_SOURCE_DIR_STIM}/src")
  foreach(prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(links libbeyond_blossom ${prop})
    if(links)
      string(REPLACE "libstim" "ae_stim" links "${links}")
      set_property(TARGET libbeyond_blossom PROPERTY ${prop} "${links}")
    endif()
  endforeach()
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_executable(bb_benchmark benchmark.cc)
target_link_libraries(bb_benchmark PRIVATE libbeyond_blossom OpenMP::OpenMP_CXX)
target_compile_options(bb_benchmark PRIVATE -O3)
target_compile_definitions(bb_benchmark PRIVATE
  BB_AE_TAILRESET=1 BB_EXPERIMENT_SHARED=1 BB_EXPERIMENT_PIPELINE=1
  BB_EXPERIMENT_OPENMP=0 BB_PMU_CACHE_LEVEL=${BB_AE_CACHE_LEVEL}
  BB_STRUCTURED_STORAGE_DIRECTIONS=12 BB_PREDECODE_SIMD=1 BB_PREDECODE_ENABLED=1
  BB_STRUCTURED_PROFILE_COUNTERS=0 BB_STRUCTURED_PROFILE_TIMERS=0 BB_STRUCTURED_PROFILE_TIMELINE=0
  BB_TAILRESET_PREDECODE_MODE=${BB_TAILRESET_PREDECODE_MODE}
  BB_TAILRESET_SYNDROME_STATS=$<BOOL:${BB_TAILRESET_SYNDROME_STATS}>)
set_target_properties(bb_benchmark PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
