cmake_minimum_required(VERSION 3.21)
project(ae_tailreset LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Offline dependencies" FORCE)
set(PYMATCHING_BUILD_PYTHON OFF CACHE BOOL "Native benchmark" FORCE)
set(PYMATCHING_BUILD_PROFILE_TOOLS OFF CACHE BOOL "Pure benchmark" FORCE)
set(PYMATCHING_PREDECODE_SIMD ON CACHE BOOL "Adopted radius scan" FORCE)
set(PYMATCHING_STRUCTURED_STORAGE_DIRECTIONS 12 CACHE STRING "Adopted layout" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_COUNTERS OFF CACHE BOOL "No profiling" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_TIMERS OFF CACHE BOOL "No profiling" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_TIMELINE OFF CACHE BOOL "No profiling" FORCE)
if(NOT IS_DIRECTORY "${PYMATCHING_SOURCE_DIR}/src" OR NOT EXISTS "${CACHED_STIM_ARCHIVE}")
  message(FATAL_ERROR "Provide PYMATCHING_SOURCE_DIR, FETCHCONTENT_SOURCE_DIR_STIM and CACHED_STIM_ARCHIVE")
endif()
add_subdirectory("${PYMATCHING_SOURCE_DIR}" pymatching EXCLUDE_FROM_ALL)
# Use the hash-verified dependency without recompiling its unchanged source.
add_library(ae_stim STATIC IMPORTED GLOBAL)
set_target_properties(ae_stim PROPERTIES IMPORTED_LOCATION "${CACHED_STIM_ARCHIVE}"
  INTERFACE_INCLUDE_DIRECTORIES "${FETCHCONTENT_SOURCE_DIR_STIM}/src")
foreach(prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
  get_target_property(links libpymatching ${prop})
  if(links)
    string(REPLACE "libstim" "ae_stim" links "${links}")
    set_property(TARGET libpymatching PROPERTY ${prop} "${links}")
  endif()
endforeach()
add_executable(tailreset_benchmark ../pmu_native/benchmark.cc)
target_compile_options(tailreset_benchmark PRIVATE -O3 -mno-avx2)
target_compile_definitions(tailreset_benchmark PRIVATE PM_AE_TAILRESET=1 PM_EXPERIMENT_SHARED=1 PM_EXPERIMENT_PIPELINE=1
  PM_PMU_CACHE_LEVEL=3 PM_SHARED_IMPORT_PLAN=0
  PM_EXPERIMENT_OPENMP=0 PM_STRUCTURED_STORAGE_DIRECTIONS=12 PM_PREDECODE_SIMD=1
  PM_PREDECODE_ENABLED=1 PM_STRUCTURED_PROFILE_COUNTERS=0 PM_STRUCTURED_PROFILE_TIMERS=0
  PM_STRUCTURED_PROFILE_TIMELINE=0 PM_PREDECODE_DIAGNOSTICS=0 PYMATCHING_ENABLE_NVTX=0
  PM_TAILRESET_PREDECODE_MODE=${PYMATCHING_TAILRESET_PREDECODE_MODE}
  PM_TAILRESET_SYNDROME_STATS=$<BOOL:${PYMATCHING_TAILRESET_SYNDROME_STATS}>)
target_link_libraries(tailreset_benchmark PRIVATE libpymatching ae_stim)
find_package(OpenMP REQUIRED COMPONENTS CXX)
target_link_libraries(tailreset_benchmark PRIVATE OpenMP::OpenMP_CXX)
set_target_properties(tailreset_benchmark PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(tailreset_policy_test policy_test.cc)
target_compile_definitions(tailreset_policy_test PRIVATE
  PM_STRUCTURED_STORAGE_DIRECTIONS=12 PM_PREDECODE_SIMD=1
  PM_TAILRESET_PREDECODE_MODE=${PYMATCHING_TAILRESET_PREDECODE_MODE}
  PM_TAILRESET_SYNDROME_STATS=$<BOOL:${PYMATCHING_TAILRESET_SYNDROME_STATS}>)
target_link_libraries(tailreset_policy_test PRIVATE libpymatching ae_stim OpenMP::OpenMP_CXX)
set_target_properties(tailreset_policy_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
