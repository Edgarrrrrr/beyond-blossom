cmake_minimum_required(VERSION 3.18)
project(pymatching_ablation_full_cycle_experiment LANGUAGES CXX)

if(NOT IS_DIRECTORY "${PYMATCHING_SOURCE_DIR}" OR NOT IS_DIRECTORY "${FETCHCONTENT_SOURCE_DIR_STIM}")
    message(FATAL_ERROR "Provide the clean PyMatching checkout and local Stim source directory")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(PYMATCHING_BUILD_TESTS OFF CACHE BOOL "Native timing build only" FORCE)
set(PYMATCHING_ENABLE_NVTX OFF CACHE BOOL "No timing instrumentation" FORCE)
set(SHARED_COMPONENT_PARALLELISM "2" CACHE STRING "Compile-time simultaneous component groups")
if(NOT SHARED_COMPONENT_PARALLELISM MATCHES "^[12]$")
    message(FATAL_ERROR "SHARED_COMPONENT_PARALLELISM must be 1 or 2")
endif()
set(PYMATCHING_SHARED_COMPONENT_PARALLELISM ${SHARED_COMPONENT_PARALLELISM} CACHE STRING "Shared component groups" FORCE)
add_compile_definitions(PM_SHARED_COMPONENT_PARALLELISM=${SHARED_COMPONENT_PARALLELISM})
set(SHARED_IMPORT_PLAN "0" CACHE STRING "Compile-time helper import-plan control")
if(NOT SHARED_IMPORT_PLAN MATCHES "^[01]$")
    message(FATAL_ERROR "SHARED_IMPORT_PLAN must be 0 or 1")
endif()
set(PYMATCHING_SHARED_IMPORT_PLAN ${SHARED_IMPORT_PLAN} CACHE BOOL "Helper import plans" FORCE)
add_compile_definitions(PM_SHARED_IMPORT_PLAN=${SHARED_IMPORT_PLAN})
set(PREDECODE_STRATEGY "0" CACHE STRING "Compile-time policy: parent/input/two-tier/minweight/leaf-only")
if(NOT PREDECODE_STRATEGY MATCHES "^[0-4]$")
    message(FATAL_ERROR "PREDECODE_STRATEGY must be 0 through 4")
endif()
set(PYMATCHING_PREDECODE_STRATEGY ${PREDECODE_STRATEGY} CACHE STRING "Compile-time policy" FORCE)
set(PREDECODE_SIMD "0" CACHE STRING "Predecode radius scan: 0 scalar, 1 AVX2 when supported")
if(NOT PREDECODE_SIMD MATCHES "^[01]$")
    message(FATAL_ERROR "PREDECODE_SIMD must be 0 or 1")
endif()
set(PYMATCHING_PREDECODE_SIMD ${PREDECODE_SIMD} CACHE BOOL "Predecode constructor binding" FORCE)
add_compile_definitions(PM_PREDECODE_SIMD=${PREDECODE_SIMD})
set(PREDECODE_ENABLED "1" CACHE STRING "Same-source predecode control")
set(STRUCTURED_PROFILE_COUNTERS "0" CACHE STRING "Separate diagnostic counters")
set(STRUCTURED_PROFILE_TIMERS "0" CACHE STRING "Separate component phase clocks")
set(STRUCTURED_PROFILE_TIMELINE "0" CACHE STRING "Separate fixed-storage phase endpoints")
foreach(flag PREDECODE_ENABLED STRUCTURED_PROFILE_COUNTERS STRUCTURED_PROFILE_TIMERS STRUCTURED_PROFILE_TIMELINE)
    if(NOT ${flag} MATCHES "^[01]$")
        message(FATAL_ERROR "${flag} must be 0 or 1")
    endif()
endforeach()
if(STRUCTURED_PROFILE_TIMELINE AND NOT STRUCTURED_PROFILE_TIMERS)
    message(FATAL_ERROR "Phase timeline requires structured phase timers")
endif()
set(PYMATCHING_PREDECODE_ENABLED ${PREDECODE_ENABLED} CACHE BOOL "Compile-time predecode control" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_COUNTERS ${STRUCTURED_PROFILE_COUNTERS} CACHE BOOL "Structured counters" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_TIMERS ${STRUCTURED_PROFILE_TIMERS} CACHE BOOL "Structured timers" FORCE)
set(PYMATCHING_STRUCTURED_PROFILE_TIMELINE ${STRUCTURED_PROFILE_TIMELINE} CACHE BOOL "Structured timeline" FORCE)
add_compile_definitions(PM_PREDECODE_ENABLED=${PREDECODE_ENABLED}
    PM_STRUCTURED_PROFILE_COUNTERS=${STRUCTURED_PROFILE_COUNTERS}
    PM_STRUCTURED_PROFILE_TIMERS=${STRUCTURED_PROFILE_TIMERS}
    PM_STRUCTURED_PROFILE_TIMELINE=${STRUCTURED_PROFILE_TIMELINE})
set(PREDECODE_CAP_CACHE "0" CACHE STRING "Explicit candidate ablation")
if(NOT PREDECODE_CAP_CACHE MATCHES "^[01]$")
    message(FATAL_ERROR "PREDECODE_CAP_CACHE must be 0 or 1")
endif()
set(PYMATCHING_PREDECODE_CAP_CACHE ${PREDECODE_CAP_CACHE} CACHE BOOL "Candidate ablation" FORCE)
add_compile_definitions(PM_PREDECODE_CAP_CACHE=${PREDECODE_CAP_CACHE})
set(PREDECODE_CAP_3X128 "0" CACHE STRING "Explicit candidate ablation")
if(NOT PREDECODE_CAP_3X128 MATCHES "^[01]$")
    message(FATAL_ERROR "PREDECODE_CAP_3X128 must be 0 or 1")
endif()
set(PYMATCHING_PREDECODE_CAP_3X128 ${PREDECODE_CAP_3X128} CACHE BOOL "Candidate ablation" FORCE)
add_compile_definitions(PM_PREDECODE_CAP_3X128=${PREDECODE_CAP_3X128})
set(PREDECODE_NEIGHBORHOOD_SIMD "0" CACHE STRING "Explicit candidate ablation")
if(NOT PREDECODE_NEIGHBORHOOD_SIMD MATCHES "^[01]$")
    message(FATAL_ERROR "PREDECODE_NEIGHBORHOOD_SIMD must be 0 or 1")
endif()
set(PYMATCHING_PREDECODE_NEIGHBORHOOD_SIMD ${PREDECODE_NEIGHBORHOOD_SIMD} CACHE BOOL "Candidate ablation" FORCE)
add_compile_definitions(PM_PREDECODE_NEIGHBORHOOD_SIMD=${PREDECODE_NEIGHBORHOOD_SIMD})
option(PREDECODE_DIAGNOSTICS "Build separate diagnostic-only executable" OFF)
if(PREDECODE_DIAGNOSTICS)
    add_compile_definitions(PM_PREDECODE_DIAGNOSTICS=1)
else()
    add_compile_definitions(PM_PREDECODE_DIAGNOSTICS=0)
endif()
option(STENCIL_GENERIC_ONLY "Compile the unchanged harness against stock upstream" OFF)
if(NOT STENCIL_GENERIC_ONLY)
    set(PYMATCHING_STRUCTURED_STORAGE_DIRECTIONS 12 CACHE STRING "Physical edge slots" FORCE)
endif()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Never fetch dependencies while collecting" FORCE)
add_compile_definitions(PYMATCHING_ENABLE_PROFILE=0 PYMATCHING_ENABLE_NVTX=0)
add_subdirectory("${PYMATCHING_SOURCE_DIR}" pymatching EXCLUDE_FROM_ALL)

if(NOT EXISTS "${CACHED_STIM_ARCHIVE}")
    message(FATAL_ERROR "Provide the verified fixed Stim archive")
endif()
add_library(experiment_stim STATIC IMPORTED GLOBAL)
set_target_properties(experiment_stim PROPERTIES IMPORTED_LOCATION "${CACHED_STIM_ARCHIVE}"
    INTERFACE_INCLUDE_DIRECTORIES "${FETCHCONTENT_SOURCE_DIR_STIM}/src")
# Keep official/core source files unchanged. Redirect only the link dependency;
# the generated local libstim target is excluded from this executable's build.
foreach(prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(current_links libpymatching ${prop})
    if(current_links)
        string(REPLACE "libstim" "experiment_stim" current_links "${current_links}")
        set_property(TARGET libpymatching PROPERTY ${prop} "${current_links}")
    endif()
endforeach()
add_executable(experiment_benchmark benchmark.cc)
option(EXPERIMENT_ENABLE_SHARED "Build shared decoder adapter when source declares it" ON)
if(EXPERIMENT_ENABLE_SHARED AND EXISTS "${PYMATCHING_SOURCE_DIR}/src/pymatching/sparse_blossom/structured/driver/shared_component_decoder.h" AND NOT STENCIL_GENERIC_ONLY)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_compile_definitions(experiment_benchmark PRIVATE PM_EXPERIMENT_SHARED=1)
else()
    target_compile_definitions(experiment_benchmark PRIVATE PM_EXPERIMENT_SHARED=0)
endif()
if(EXISTS "${PYMATCHING_SOURCE_DIR}/src/pymatching/sparse_blossom/structured/driver/openmp_component_decoder.h" AND NOT STENCIL_GENERIC_ONLY)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_compile_definitions(experiment_benchmark PRIVATE PM_EXPERIMENT_OPENMP=1)
else()
    target_compile_definitions(experiment_benchmark PRIVATE PM_EXPERIMENT_OPENMP=0)
endif()
set(EXPERIMENT_HAS_PIPELINE 0)
if(EXPERIMENT_ENABLE_SHARED AND NOT STENCIL_GENERIC_ONLY)
    set(shared_header "${PYMATCHING_SOURCE_DIR}/src/pymatching/sparse_blossom/structured/driver/shared_component_decoder.h")
    if(EXISTS "${shared_header}")
        file(READ "${shared_header}" shared_header_text)
        if(shared_header_text MATCHES "class DoubleBufferComponentDecoder")
            set(EXPERIMENT_HAS_PIPELINE 1)
        endif()
    endif()
endif()
target_compile_definitions(experiment_benchmark PRIVATE PM_EXPERIMENT_PIPELINE=${EXPERIMENT_HAS_PIPELINE})
target_compile_definitions(experiment_benchmark PRIVATE PM_PREDECODE_STRATEGY=${PREDECODE_STRATEGY})
if(STENCIL_GENERIC_ONLY)
    target_compile_definitions(experiment_benchmark PRIVATE STENCIL_GENERIC_ONLY=1)
else()
    target_compile_definitions(experiment_benchmark PRIVATE PM_STRUCTURED_STORAGE_DIRECTIONS=12)
endif()
target_include_directories(experiment_benchmark PRIVATE "${PYMATCHING_SOURCE_DIR}/src")
target_compile_options(experiment_benchmark PRIVATE -O3 -mno-avx2)
target_link_libraries(experiment_benchmark PRIVATE libpymatching experiment_stim)
target_link_options(experiment_benchmark PRIVATE -pthread -O3)
if(TARGET OpenMP::OpenMP_CXX)
    target_link_libraries(experiment_benchmark PRIVATE OpenMP::OpenMP_CXX)
endif()
set_target_properties(experiment_benchmark PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
